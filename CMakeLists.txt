cmake_minimum_required(VERSION 3.20)
project(parallel_file_system VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Protobuf CONFIG REQUIRED)
find_package(gRPC CONFIG REQUIRED)
set(PROTO ${CMAKE_CURRENT_SOURCE_DIR}/proto/pfs.proto)
set(GEN_DIR ${CMAKE_CURRENT_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${GEN_DIR})
add_custom_command(OUTPUT ${GEN_DIR}/pfs.pb.cc ${GEN_DIR}/pfs.pb.h ${GEN_DIR}/pfs.grpc.pb.cc ${GEN_DIR}/pfs.grpc.pb.h
  COMMAND protobuf::protoc --proto_path=${CMAKE_CURRENT_SOURCE_DIR}/proto --cpp_out=${GEN_DIR} --grpc_out=${GEN_DIR} --plugin=protoc-gen-grpc=$<TARGET_FILE:gRPC::grpc_cpp_plugin> ${PROTO}
  DEPENDS ${PROTO})
add_library(pfs_proto ${GEN_DIR}/pfs.pb.cc ${GEN_DIR}/pfs.grpc.pb.cc)
target_include_directories(pfs_proto PUBLIC ${GEN_DIR})
target_link_libraries(pfs_proto PUBLIC protobuf::libprotobuf gRPC::grpc++)
add_library(pfs_client src/client.cpp)
target_include_directories(pfs_client PUBLIC include)
target_link_libraries(pfs_client PUBLIC pfs_proto)
foreach(target coordinator storage)
  add_executable(pfs-${target} src/${target}_server.cpp)
  target_include_directories(pfs-${target} PRIVATE include)
  target_link_libraries(pfs-${target} PRIVATE pfs_proto)
endforeach()
add_executable(pfsctl src/cli.cpp)
target_include_directories(pfsctl PRIVATE include)
target_link_libraries(pfsctl PRIVATE pfs_client)
enable_testing()
add_executable(integration_test tests/integration_test.cpp)
target_link_libraries(integration_test PRIVATE pfs_client)
add_test(NAME unit_cli_usage COMMAND pfsctl)
set_tests_properties(unit_cli_usage PROPERTIES WILL_FAIL TRUE)
add_executable(stress_test tests/stress_test.cpp)
target_link_libraries(stress_test PRIVATE pfs_client)
install(TARGETS pfs-coordinator pfs-storage pfsctl RUNTIME DESTINATION bin)
